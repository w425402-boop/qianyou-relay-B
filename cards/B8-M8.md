## 3. 定桩说明（本棒要摘掉的桩）

开局两个函数是 `QY.seam.stub('M8 QY.toggleActivityStatus')` 形态，一调用即抛 `[SEAM 未实现]`。

**交付标准**：M8 的 2 个契约函数全部不再是桩，`node tests/contract.js` 的「本棒模块仍是桩」检查转绿。

```js
// js/modules/admin.js —— 从桩到真实现（缩短版骨架，够联调用即可）
(function (root) {
  var QY = (root.QY = root.QY || {});
  var filter = { status: 'all', keyword: '' };          // 面板内部状态，不进 store

  // 契约函数：切状态并落盘。nextStatus 非法则取反当前值
  QY.toggleActivityStatus = function (activityId, nextStatus) {
    var state = QY.store.load(), act = null;
    for (var i = 0; i < state.activities.length; i++) if (state.activities[i].id === activityId) act = state.activities[i];
    if (!act) return { ok: false, reason: '未找到活动' };
    act.status = (nextStatus === 'ongoing' || nextStatus === 'ended')
      ? nextStatus
      : (act.status === 'ended' ? 'ongoing' : 'ended');
    QY.store.save(state);                               // 整份 state 落盘
    QY.renderAdmin();                                   // 立即重渲染，按钮文字跟着变
    return { ok: true, activity: act };
  };

  // 渲染：一列活动 + 每行一个切换按钮 + 顶部两个统计数（进行中/已结束）
  QY.renderAdmin = function () {
    var main = document.getElementById('app-main');
    if (!main) return;                                  // 容器缺失静默返回
    var acts = QY.store.load().activities || [];
    // 每行：标题 / 分类 / joined-capacity / 状态标签 / 「${已结束?'恢复':'结束'}」按钮
  };
})(typeof window !== 'undefined' ? window : globalThis);

// js/modules/integration.js —— 联调胶水（只做"接线"，不改别人逻辑）
// 例：确保顶部栏有 #/admin 入口、底部 Tab 高亮与实际路由对齐、缺失的 hash 兜底到发现页
```

两个坑：①切状态后必须重渲染，否则按钮文字与 store 不一致，你截图取证时会被自己骗；②`#/admin` 不占底部固定 4 项 Tab（契约第 2 节），入口放在顶部栏或"我的"页，别硬塞进 Tab 栏。
