## 1. 你这一棒交付什么

**为什么拆给你**：考察在不修改 M4 详情页文件的前提下完成跨模块路由接线、未读消息状态机与会话持久化。

逐条交付，不许减项：

1. **会话列表**：活动群聊（每个已报名活动一个，标题为活动标题）与私信（与发起人 1 对 1，标题为对方昵称）；显示最后消息、时间、未读数字角标。

4. **未读**：打开会话后 `unread` 清零；M6 自己往 store 里预置 **2 条未读消息与种子会话**（写进交接单，B7/B8 联调要用）。
5. **接线**：把 M4 详情页"联系发起人"接到私信、"进群聊"接到活动群聊；支持 `#/chat?conv=:id` 直接定位指定会话。

**交付物**：`js/modules/chat.js`；`tests/check.js` 里追加 2 条断言（发消息后 messages 长度+1、打开会话后 unread 为 0）；《交接单》。

## 2. 契约前置（先看契约再写代码，10 分钟）

对照 CONTRACT.md 第 4 节，本棒 3 个对外函数：

| 函数 | 入参 | 返回值 | 出错行为 |
| :--- | :--- | :--- | :--- |
| `QY.renderChat(params)` | `Object`（含 `conv`，可缺省） | 渲染会话列表或指定会话，返回 `undefined` | `conv` 指向不存在的会话 → 渲染列表并提示，不抛 |
| `QY.openConversation(convId)` | `string` | 该会话对象（`unread` 已清零并落盘） | 会话不存在 → `null` |
| `QY.sendMessage(convId, text)` | `string, string` | 新消息对象 `{ id, convId, fromId, text, time }` | 空/纯空白 `text` → `null` 且**不写库**；会话不存在 → `null` |

会话 id 规则（契约 1.5）：群聊 `grp_<actId>`，私信 `dm_<userId>`。`sendMessage` 必须同时更新会话的 `lastMsg` / `lastTime`，否则会话列表停在旧消息。

**只许改（owns）**：`js/modules/chat.js`
**只许追加（appendOnly）**：`css/style.css`（只追加 M6 区块）、`tests/check.js`（只追加 M6 断言）
**不在你手里**：`js/modules/detail.js`（M4 owns）、`js/router.js`（M1 owns）—— 接线靠**事件委托 + 改 `location.hash`**，一个字节都不许写进那两个文件

红线：**不遵守就是契约违规，扣 3 分/条。**

## 3. 定桩说明（本棒要摘掉的桩）

开局三个函数是 `QY.seam.stub('M6 QY.openConversation')` 形态，一调用即抛 `[SEAM 未实现]`。

**交付标准**：M6 的 3 个契约函数全部不再是桩，`node tests/contract.js` 的「本棒模块仍是桩」检查转绿。

```js
// js/modules/chat.js —— 从桩到真实现（会话引导 + 未读状态机是重点）
(function (root) {
  var QY = (root.QY = root.QY || {}), ME = 'u01';
  // 首次进页面时补种子会话：2 条未读（他人发的）+ 各已报名活动的群聊
  function ensureSeed() {
    var state = QY.store.load();
    if (state.conversations && state.conversations.length) return state;
    var convs = []; (state.activities || []).forEach(function (a) {
      if ((a.signedUserIds || []).indexOf(ME) > -1) {
        convs.push({ id: 'grp_' + a.id, type: 'group', title: a.title, activityId: a.id, members: [ME, a.hostId],
                     lastMsg: '周五不见不散！', lastTime: new Date().toISOString(), unread: 2 });   // 预置 2 条未读
      }
    });
    state.conversations = convs;
    state.messages = [{ id: 'm01', convId: convs[0] && convs[0].id, fromId: 'u04', text: '周五不见不散！', time: new Date().toISOString() }];
    QY.store.save(state);
    return state;
  }

  // 契约函数：打开会话 → 未读清零 → 落盘。顺序不能反
  QY.openConversation = function (convId) {
    var state = ensureSeed();
    for (var i = 0; i < state.conversations.length; i++) {
      if (state.conversations[i].id !== convId) continue;
      state.conversations[i].unread = 0;
      QY.store.save(state);                       // 先清零再返回，刷新才不反弹
      return state.conversations[i];
    }
    return null;
  };

  // 契约函数：发消息。空白 → null 且不写库；同时刷新会话摘要
  QY.sendMessage = function (convId, text) {
    var t = String(text == null ? '' : text).trim();
    if (!t) return null;
    var state = ensureSeed(), conv = null;
    for (var i = 0; i < state.conversations.length; i++) if (state.conversations[i].id === convId) conv = state.conversations[i];
    if (!conv) return null;
    var msg = { id: 'm' + Date.now(), convId: convId, fromId: ME, text: t, time: new Date().toISOString() };
    state.messages.push(msg);
    conv.lastMsg = t; conv.lastTime = msg.time;                     // 会话摘要同步，列表才不落后
    QY.store.save(state); QY.renderChat({ conv: convId });           // 重渲染并滚到底
    return msg;
  };
  // 接线：事件委托挂在 document 上，绝不改 detail.js / router.js
  document.addEventListener('click', function (e) {
    var el = e.target.closest ? e.target.closest('[data-chat-target]') : null; if (!el) return;
    var kind = el.getAttribute('data-chat-target');
    location.hash = kind === 'dm' ? '#/chat?conv=dm_' + (el.getAttribute('data-host-id') || 'u04') : '#/chat?conv=grp_' + (el.getAttribute('data-activity-id') || '');
  });
  QY.renderChat = function (params) { /* 列表 / 气泡左右分布 / 输入框 / 滚到底 */ };
})(typeof window !== 'undefined' ? window : globalThis);
```

三个坑：①`#/chat?conv=` 的 query 自己从 `location.hash` 里解析（`router.js` 不帮你拆）；②种子会话必须在首次加载时补齐，且**只在 conversations 为空时补**，否则每次进页面都把用户已读状态覆盖成 2；③气泡左右靠 `fromId === ME` 判断，别用"数组最后一条"，历史消息一多就错。
