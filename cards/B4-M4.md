对照 CONTRACT.md 第 4 节，本棒 3 个对外函数：

| 函数 | 入参 | 返回值 | 出错行为 |
| :--- | :--- | :--- | :--- |
| `QY.renderDetail(id)` | `string` | 渲染 `#app-main` 内详情视图，返回 `undefined` | id 不存在 → 渲染"活动不存在"空态，**不抛** |

**只许追加（appendOnly）**：`css/style.css`（只追加 M4 区块）、`tests/check.js`（只追加 M4 断言）
**不属于你**：`js/router.js`（M1 owns）—— 路由注册已由 M1 做完，你只实现 `QY.routes.detail` 调用的渲染函数

## 3. 定桩说明（本棒要摘掉的桩）

开局三个函数是 `QY.seam.stub('M4 QY.renderDetail')` 形态，一调用即抛 `[SEAM 未实现]`。

**交付标准**：M4 的 3 个契约函数全部不再是桩，`node tests/contract.js` 的「本棒模块仍是桩」检查转绿。

```js
// js/modules/detail.js —— 从桩到真实现
(function (root) {
  var QY = (root.QY = root.QY || {});
  var ME = 'u01';

  // 取活动 + 取整份 state：报名类操作必须拿整份 state 再 save，不能只存单条
  function findAct(state, id) {
    for (var i = 0; i < state.activities.length; i++) if (state.activities[i].id === id) return state.activities[i];
    return null;
  }

  // 契约函数：报名。四种结局都要有确定返回值，check.js 逐条断言
  QY.signup = function (activityId) {
    var state = QY.store.load();
    var act = findAct(state, activityId);
    if (!act) return { ok: false, reason: '未找到活动' };
    if (act.status === 'ended') return { ok: false, reason: '活动已结束' };
    if (!Array.isArray(act.signedUserIds)) act.signedUserIds = [];
    if (act.signedUserIds.indexOf(ME) > -1) return { ok: true, joined: act.joined };   // 幂等
    if (act.joined >= act.capacity) return { ok: false, reason: '人数已满' };
    act.signedUserIds.push(ME);
    act.joined = act.signedUserIds.length;
    QY.store.save(state);
    QY.renderDetail(activityId);
    return { ok: true, joined: act.joined };
  };

  // 契约函数：取消报名。未报名就返回失败，不硬删
  QY.cancelSignup = function (activityId) {
    var state = QY.store.load();
    var act = findAct(state, activityId);
    if (!act) return { ok: false, reason: '未找到活动' };
    var idx = (act.signedUserIds || []).indexOf(ME);
    if (idx === -1) return { ok: false, reason: '未报名' };
    act.signedUserIds.splice(idx, 1);
    act.joined = Math.max(0, act.signedUserIds.length);
    QY.store.save(state);
    QY.renderDetail(activityId);
    return { ok: true, joined: act.joined };
  };

  // 渲染：8 项信息 + 参与者列表 + 评论（time 升序）+ 两个预留按钮
  QY.renderDetail = function (id) { /* 全部走 QY.store.getActivity(id) */ };
})(typeof window !== 'undefined' ? window : globalThis);
```

两个坑：①`QY.store.save(state)` 必须传整份 state，只传活动会把 users/messages 洗掉，B6 的会话当场消失；②评论 id 用 `'cm' + Date.now()`，同一毫秒连发两条会撞 id，加个序号后缀。
